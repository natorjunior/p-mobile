📦 Variáveis: var vs let

var - Mutável 

// Pode ser alterada
var nome = "Maria"
nome = "Maria Silva"  // ✅ OK

idade = 26           // ✅ OK
let - Imutável 

// NÃO pode ser alterada

⚠️ Regra de ouro: Sempre use let por padrão. Use var apenas quando realmente precisar alterar o valor.

Structs e Classes

Modelando dados no Swift

📦 Structs (Estruturas)


// 📌 Definindo uma struct
struct Aluno {
    var nome: String
    var idade: Int
    var curso: String
    
    // Método
    func apresentar() -> String {
        return "Sou \(nome), tenho \(idade) anos"
    }
    
    // Método que modifica (precisa de mutating)
    mutating func fazerAniversario() {
        idade += 1
    }
}

// 📌 Usando a struct
var joao = Aluno(nome: "João", idade: 20, curso: "ADS")
print(joao.apresentar())  // "Sou João, tenho 20 anos"

joao.fazerAniversario()
print(joao.idade)  // 21
💡 Structs são VALUE TYPES: São copiadas quando atribuídas a outra variável.

🏛️ Classes


// 📌 Definindo uma classe
class Professor {
    var nome: String
    var disciplina: String
    
    // Inicializador obrigatório
    init(nome: String, disciplina: String) {
        self.nome = nome
        self.disciplina = disciplina
    }
    
    func apresentar() -> String {
        return "Prof. \(nome) - \(disciplina)"
    }
    
    // Deinicializador (quando objeto é destruído)
    deinit {
        print("\(nome) foi removido da memória")
    }
}

// 📌 Usando a classe
let maria = Professor(nome: "Maria", disciplina: "iOS")
print(maria.apresentar())
⚠️ Classes são REFERENCE TYPES: Variáveis apontam para o mesmo objeto na memória.

⚖️ Struct vs Class

📦 Struct

  • Value Type (copiado)
  • Inicializador automático
  • Sem herança
  • Mais rápido e seguro
  • Ideal para dados simples

var a = Aluno(nome: "A", idade: 20)
var b = a  // Cópia!
b.nome = "B"
print(a.nome)  // "A" ✅

🏛️ Class

  • Reference Type (referência)
  • Precisa de init()
  • Suporta herança
  • Tem deinit
  • Ideal para objetos complexos

var a = Professor(nome: "A", disc: "iOS")
var b = a  // Mesma referência!
b.nome = "B"
print(a.nome)  // "B" ⚠️

❓ Optionals - O poder do nil seguro


// 📌 Optional = pode ter valor OU ser nil
var email: String? = nil       // Não tem valor ainda
var telefone: String? = "123"  // Tem valor

// 📌 Unwrapping seguro com if let
if let emailValido = email {
    print("Email: \(emailValido)")
} else {
    print("Email não informado")
}

// 📌 Guard let (saída antecipada)
func enviarEmail(para email: String?) {
    guard let emailValido = email else {
        print("Email inválido!")
        return
    }
    print("Enviando para \(emailValido)")
}

// 📌 Nil coalescing (??)
let nome: String? = nil
let saudacao = "Olá, \(nome ?? "Visitante")"  // "Olá, Visitante"

🔗 Optional Chaining


struct Endereco {
    var rua: String
    var numero: Int
}

struct Pessoa {
    var nome: String
    var endereco: Endereco?  // Pode não ter endereço
}

let joao = Pessoa(nome: "João", endereco: nil)
let maria = Pessoa(
    nome: "Maria", 
    endereco: Endereco(rua: "Av. Brasil", numero: 100)
)

// 📌 Optional chaining com ?
print(joao.endereco?.rua)   // nil (não causa erro!)
print(maria.endereco?.rua)  // Optional("Av. Brasil")

// 📌 Combinando com ?? 
let ruaJoao = joao.endereco?.rua ?? "Não informado"
print(ruaJoao)  // "Não informado"
💡 Optionals previnem crashes! Ao invés de acessar nil e crashar, Swift te obriga a tratar o caso.

🎁 Closures (Funções Anônimas)


// 📌 Closure básica
let saudacao = { (nome: String) -> String in
    return "Olá, \(nome)!"
}
print(saudacao("João"))  // "Olá, João!"

// 📌 Closure como parâmetro (muito usado em SwiftUI!)
let numeros = [3, 1, 4, 1, 5, 9, 2, 6]

// Ordenar com closure
let ordenados = numeros.sorted { $0 < $1 }
print(ordenados)  // [1, 1, 2, 3, 4, 5, 6, 9]

// Map, Filter, Reduce
let dobrados = numeros.map { $0 * 2 }           // [6, 2, 8, ...]
let pares = numeros.filter { $0 % 2 == 0 }      // [4, 2, 6]
let soma = numeros.reduce(0) { $0 + $1 }        // 31

// 📌 Trailing closure syntax
DispatchQueue.main.async {
    print("Executando na main thread")
}

🔄 @State - Gerenciando Estado


struct ContadorView: View {
    @State private var contador = 0  // 👈 Estado reativo
    
    var body: some View {
        VStack(spacing: 20) {
            Text("Contador: \(contador)")
                .font(.largeTitle)
            
            HStack(spacing: 20) {
                Button("−") {
                    contador -= 1  // Muda o estado
                }
                .buttonStyle(.bordered)
                
                Button("+") {
                    contador += 1  // UI atualiza automaticamente!
                }
                .buttonStyle(.borderedProminent)
            }
        }
        .padding()
    }
}
💡 @State marca uma propriedade como "fonte da verdade". Quando muda, SwiftUI re-renderiza a View automaticamente!

📝 TextField e @Binding


struct FormularioView: View {
    @State private var nome = ""
    @State private var email = ""
    @State private var aceitouTermos = false
    
    var body: some View {
        Form {
            Section("Dados Pessoais") {
                TextField("Nome completo", text: $nome)
                TextField("E-mail", text: $email)
                    .keyboardType(.emailAddress)
                    .autocapitalization(.none)
            }
            
            Section {
                Toggle("Aceito os termos", isOn: $aceitouTermos)
            }
            
            Section {
                Button("Cadastrar") {
                    print("Nome: \(nome), Email: \(email)")
                }
                .disabled(!aceitouTermos || nome.isEmpty)
            }
        }
    }
}

Parte 4

UIKit

O framework clássico do iOS

📱 UIKit - Quando usar?

✅ Use UIKit quando:

  • Projeto legado existente
  • Precisa de controle pixel-perfect
  • Funcionalidades não disponíveis no SwiftUI
  • Suporte a iOS < 13

✅ Use SwiftUI quando:

  • Projeto novo
  • Prototipagem rápida
  • UI reativa e moderna
  • Multiplataforma Apple
💡 Na prática: Muitos apps usam os dois! SwiftUI pode incorporar views UIKit e vice-versa.

🏗️ UIKit Básico


import UIKit

class ViewController: UIViewController {
    
    // Elementos de UI
    let label = UILabel()
    let button = UIButton(type: .system)
    
    override func viewDidLoad() {
        super.viewDidLoad()
        view.backgroundColor = .white
        
        // Configurar Label
        label.text = "Olá, UIKit!"
        label.font = .systemFont(ofSize: 24, weight: .bold)
        label.textAlignment = .center
        
        // Configurar Button
        button.setTitle("Clique aqui", for: .normal)
        button.addTarget(self, action: #selector(botaoClicado), 
                         for: .touchUpInside)
        
        // Adicionar à view
        setupLayout()
    }
    
    @objc func botaoClicado() {
        label.text = "Você clicou! 🎉"
    }
}

📐 Auto Layout (Constraints)


func setupLayout() {
    // Desabilitar autoresizing mask
    label.translatesAutoresizingMaskIntoConstraints = false
    button.translatesAutoresizingMaskIntoConstraints = false
    
    // Adicionar subviews
    view.addSubview(label)
    view.addSubview(button)
    
    // Definir constraints
    NSLayoutConstraint.activate([
        // Label centralizada horizontalmente
        label.centerXAnchor.constraint(equalTo: view.centerXAnchor),
        label.centerYAnchor.constraint(equalTo: view.centerYAnchor, 
                                        constant: -50),
        
        // Button abaixo da label
        button.centerXAnchor.constraint(equalTo: view.centerXAnchor),
        button.topAnchor.constraint(equalTo: label.bottomAnchor, 
                                     constant: 20),
        button.widthAnchor.constraint(equalToConstant: 200),
        button.heightAnchor.constraint(equalToConstant: 44)
    ])
}

// 📌 Passo 2: Gerenciador de tarefas
import SwiftUI

@Observable
class TarefasViewModel {
    var tarefas: [Tarefa] = []
    var novaTarefa = ""
    var novaDescricao = ""
    
    func adicionar() {
        guard !novaTarefa.isEmpty else { return }
        let tarefa = Tarefa(titulo: novaTarefa, descricao: novaDescricao)
        tarefas.append(tarefa)
        novaTarefa = ""
        novaDescricao = ""
    }
    
    func remover(at offsets: IndexSet) {
        tarefas.remove(atOffsets: offsets)
    }
    
    func alternarConclusao(_ tarefa: Tarefa) {
        if let index = tarefas.firstIndex(where: { $0.id == tarefa.id }) {
            tarefas[index].concluida.toggle()
        }
    }
}

                    TextField("Título da tarefa", text: $viewModel.novaTarefa)
                    TextField("Descrição (opcional)", 
                              text: $viewModel.novaDescricao)
                }
            }
            .navigationTitle("Nova Tarefa")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .cancellationAction) {
                    Button("Cancelar") { isPresented = false }
                }
                ToolbarItem(placement: .confirmationAction) {
                    Button("Salvar") {
                        viewModel.adicionar()
                        isPresented = false
                    }
                    .disabled(viewModel.novaTarefa.isEmpty)
                }
            }
        }
    }
