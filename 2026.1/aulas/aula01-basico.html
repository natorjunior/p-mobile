Classe simples


class Aluno {
  constructor(nome, turma) {
    this.nome = nome;
    this.turma = turma;
  }

  saudacao() {
    return `Olá, eu sou ${this.nome} da turma ${this.turma}`;
  }
}

const maria = new Aluno("Maria", "2026.1");
console.log(maria.saudacao());
