Entendendo o Problema

Como compartilhar dados entre componentes
sem passar props manualmente por vários níveis?

🔄 Relembrando: Props

// Componente Pai
const ListaDeTarefas = () => {
  const tarefas = []; // dados no pai
  return <ExibirTarefa tarefas={tarefas} />;
};

🔄 Relembrando: Props

// Componente Pai
const ListaDeTarefas = () => {
  const tarefas = [
    { id: 1, texto: "Estudar React", concluida: false },
    { id: 2, texto: "Fazer exercícios", concluida: true }
  ];
  return <ExibirTarefa tarefas={tarefas} />;
};

// Componente Filho recebe via props
const ExibirTarefa = ({ tarefas }) => {
  return (
    <View>
      {tarefas.map(t => (
        <Tarefa key={t.id} tarefa={t} />
      ))}
    </View>
  );
};
📌 Props fluem de cima para baixo (unidirecional)

⚠️ O Problema: Prop Drilling

Quando dados precisam passar por muitos níveis de componentes: 

// Dados em App
<App user={usuario}>
  <Header user={usuario}>            {/* passa user */}
    <Profile user={usuario}>         {/* passa user */}
      <Avatar user={usuario} />      {/* finalmente usa! */}
    </Profile>
  </Header>
</App>
❌ Ineficiente: Componentes intermediários não usam o dado, apenas repassam

Context API

Uma forma de compartilhar estado sem passar props por todos os componentes

✨ Solução: useContext

// Passo 1: Criar Context
import { createContext } from 'react';
const UserContext = createContext();

✨ Solução: useContext

// Passo 2: Criar Provider
const UserProvider = ({ children }) => {
  const usuario = { nome: 'João', idade: 25 };
  
  return (
    <UserContext.Provider value={usuario}>
      {children}
    </UserContext.Provider>
  );
};

✨ Solução: useContext

// Passo 3: Envolver App com Provider
const App = () => {
  return (
    <UserProvider>
      <Header />
      <Content />
      <Footer />
    </UserProvider>
  );
};
✅ Sem prop drilling! Todos os componentes filhos podem usar dados

✨ Solução: useContext

// Passo 4: Consumir dados em qualquer componente
import { useContext } from 'react';

const Avatar = () => {
  const user = useContext(UserContext);
  
  return (
    <View>
      <Text>{user.nome}</Text>
      <Text>{user.idade} anos</Text>
    </View>
  );
};

Nenhuma prop intermediária! 🎉

O que é Context API? 🤔

  • Forma de compartilhar dados entre componentes
  • Evita "prop drilling" (passar props em vários níveis)
  • Ideal para dados globais: temas, usuário logado, idioma

Atividade 1: Sistema de Tema Simples 🎨

Crie um aplicativo com:

  • Um ThemeContext para gerenciar tema claro/escuro
  • Um botão para alternar entre temas
  • Dois componentes que usam o tema atual
  • Cores diferentes para cada tema

Solução: Sistema de Tema 🎨


    // ThemeContext.js
    import { createContext } from 'react';
    export const ThemeContext = createContext();

    // App.js
    const App = () => {
      const [theme, setTheme] = useState('light');
      return (
        <ThemeContext.Provider value={{ theme, setTheme }}>
      <Header />
      <Content />
        </ThemeContext.Provider>
      );
    };

    // Header.js
    const Header = () => {
      const { theme, setTheme } = useContext(ThemeContext);
      return (
        <View style={{
      backgroundColor: theme === 'light' ? '#fff' : '#333',
      padding: 20
        }}>
      <Button
        title={theme === 'light' ? '🌙' : '☀️'}
        onPress={() => setTheme(theme === 'light' ? 'dark' : 'light')}
      />
        </View>
      );
    };

    // Content.js
    const Content = () => {
      const { theme } = useContext(ThemeContext);
      return (
        <View style={{
      flex: 1,
      backgroundColor: theme === 'light' ? '#f0f0f0' : '#222',
      padding: 20
        }}>
      <Text style={{ color: theme === 'light' ? '#000' : '#fff' }}>
        Conteúdo do App
      </Text>
        </View>
      );
    };

Atividade 2: Contador Global 🔢

Desenvolva:

  • Um CounterContext para manter um contador global
  • Botões para incrementar e decrementar em um componente
  • Exibir o valor atual em outro componente
  • Adicionar um botão de reset

Solução: Contador Global 🔢


    // CounterContext.js
    import { createContext } from 'react';
    export const CounterContext = createContext();

    // App.js
    const App = () => {
      const [count, setCount] = useState(0);
      return (
        <CounterContext.Provider value={{ count, setCount }}>
      <Controls />
      <Display />
        </CounterContext.Provider>
      );
    };

    // Controls.js
    const Controls = () => {
      const { count, setCount } = useContext(CounterContext);
      return (
        <View style={styles.controls}>
      <Button title="-" onPress={() => setCount(count - 1)} />
      <Button title="Reset" onPress={() => setCount(0)} />
      <Button title="+" onPress={() => setCount(count + 1)} />
        </View>
      );
    };

    // Display.js
    const Display = () => {
      const { count } = useContext(CounterContext);
      return (
        <View style={styles.display}>
      <Text style={styles.number}>{count}</Text>
        </View>
      );
    };

Projeto Prático: Todo List com Temas 📝

O que vamos criar:

  • Lista de tarefas com tema customizável
  • Adicionar e remover tarefas
  • Alternar entre tema claro e escuro
  • Usar Context para gerenciar estado global

Passo 1: Criando os Contexts


// contexts/ThemeContext.js
import { createContext } from 'react';

export const ThemeContext = createContext();

// contexts/TodoContext.js
import { createContext } from 'react';

export const TodoContext = createContext();

Passo 2: Criando os Providers


// App.js
import React, { useState } from 'react';
import { ThemeContext, TodoContext } from './contexts';

const App = () => {
  const [theme, setTheme] = useState('light');
  const [todos, setTodos] = useState([]);

  return (
    <ThemeContext.Provider value={{ theme, setTheme }}>
      <TodoContext.Provider value={{ todos, setTodos }}>
        <MainApp />
      </TodoContext.Provider>
    </ThemeContext.Provider>
  );
};

Passo 3: Componente TodoInput


// components/TodoInput.js
import React, { useState, useContext } from 'react';
import { View, TextInput, Button, StyleSheet } from 'react-native';
import { TodoContext, ThemeContext } from '../contexts';

const TodoInput = () => {
  const [text, setText] = useState('');
  const { todos, setTodos } = useContext(TodoContext);
  const { theme } = useContext(ThemeContext);

  const addTodo = () => {
    if (text.trim()) {
      setTodos([...todos, { id: Date.now(), text, completed: false }]);
      setText('');
    }
  };

  return (
    <View style={styles.container}>
      <TextInput
        style={[
          styles.input,
          { backgroundColor: theme === 'light' ? '#fff' : '#333',
            color: theme === 'light' ? '#000' : '#fff' }
        ]}
        value={text}
        onChangeText={setText}
        placeholder="Nova tarefa..."
        placeholderTextColor={theme === 'light' ? '#666' : '#999'}
      />
      <Button title="Adicionar" onPress={addTodo} />
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    padding: 20,
    flexDirection: 'row',
  },
  input: {
    flex: 1,
    marginRight: 10,
    padding: 10,
    borderWidth: 1,
    borderRadius: 5,
  },
});

Passo 4: Componente TodoList


// components/TodoList.js
import React, { useContext } from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';
import { TodoContext, ThemeContext } from '../contexts';

const TodoList = () => {
  const { todos, setTodos } = useContext(TodoContext);
  const { theme } = useContext(ThemeContext);

  const toggleTodo = (id) => {
    setTodos(todos.map(todo =>
      todo.id === id ? { ...todo, completed: !todo.completed } : todo
    ));
  };

  const deleteTodo = (id) => {
    setTodos(todos.filter(todo => todo.id !== id));
  };

  return (
    <View style={styles.container}>
      {todos.map(todo => (
        <TouchableOpacity
          key={todo.id}
          style={[
            styles.todoItem,
            { backgroundColor: theme === 'light' ? '#f8f8f8' : '#333' }
          ]}
          onPress={() => toggleTodo(todo.id)}
          onLongPress={() => deleteTodo(todo.id)}
        >
          <Text style={[
            styles.todoText,
            { 
              color: theme === 'light' ? '#000' : '#fff',
              textDecorationLine: todo.completed ? 'line-through' : 'none'
            }
          ]}>
            {todo.text}
          </Text>
        </TouchableOpacity>
      ))}
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    padding: 20,
  },
  todoItem: {
    padding: 15,
    marginBottom: 10,
    borderRadius: 5,
    elevation: 2,
  },
  todoText: {
    fontSize: 16,
  },
});

Passo 5: Componente ThemeToggle


// components/ThemeToggle.js
import React, { useContext } from 'react';
import { View, Switch, Text, StyleSheet } from 'react-native';
import { ThemeContext } from '../contexts';

const ThemeToggle = () => {
  const { theme, setTheme } = useContext(ThemeContext);

  return (
    <View style={styles.container}>
      <Text style={{ color: theme === 'light' ? '#000' : '#fff' }}>
        {theme === 'light' ? '☀️' : '🌙'}
      </Text>
      <Switch
        value={theme === 'dark'}
        onValueChange={(value) => setTheme(value ? 'dark' : 'light')}
      />
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    flexDirection: 'row',
    alignItems: 'center',
    justifyContent: 'flex-end',
    padding: 20,
  },
});

Passo 6: Juntando Tudo


// MainApp.js
import React, { useContext } from 'react';
import { View, StyleSheet } from 'react-native';
import { ThemeContext } from './contexts';
import { TodoInput, TodoList, ThemeToggle } from './components';

const MainApp = () => {
  const { theme } = useContext(ThemeContext);

  return (
    <View style={[
      styles.container,
      { backgroundColor: theme === 'light' ? '#fff' : '#121212' }
    ]}>
      <ThemeToggle />
      <TodoInput />
      <TodoList />
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    flex: 1,
  },
});

Benefícios do Context 🎯

  • Código mais limpo e organizado
  • Evita prop drilling
  • Estado global facilmente acessível
  • Separação clara de responsabilidades
  • Facilita manutenção e escalabilidade

Considerações Finais 🤔

  • Use Context para estado global
  • Evite criar muitos contextos
  • Combine com outros hooks quando necessário
  • Considere performance em apps grandes
