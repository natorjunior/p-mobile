💻 Na Prática: Cadastro de Usuário

🎨 Tela (Compose) 

@Composable
fun RegistroScreen(vm: RegistroVM) {
    var nome by remember { 
        mutableStateOf("") 
    }
    
    TextField(value = nome,
        onValueChange = { nome = it })
    
    Button(onClick = { 
        vm.salvarUsuario(nome)
    }) {
        Text("Cadastrar")
    }
}
⚙️ Lógica (ViewModel) 

class RegistroVM(
    val repo: UsuarioRepo
) : ViewModel() {

    fun salvarUsuario(nome: String) {
        if (nome.isBlank()) return
        
        val user = Usuario(
            nome = nome,
            data = LocalDate.now()
        )
        repo.salvar(user)
    }
}
📌 Observe: A tela só sabe chamar o ViewModel. O ViewModel só sabe chamar o Repository. Ninguém faz tudo sozinho!

☕ Por que Kotlin e não Java?

Kotlin 

// Classe de dados em 1 linha!
data class Pessoa(
    val nome: String,
    val idade: Int
)

// Null safety nativo
val nome: String? = null
println(nome?.length ?: 0)

// Funções de extensão
fun String.grito() = this.uppercase() + "!"
println("olá".grito()) // "OLÁ!"
Java 

// Muito mais código...
public class Pessoa {
    private String nome;
    private int idade;
    // getters, setters, equals,
    // hashCode, toString...
    // +50 linhas de código!
}

// NullPointerException 💥
String nome = null;
System.out.println(nome.length());

Kotlin é 100% interoperável com Java! 🤝

📦 Variáveis: val vs var

val - Imutável 

// NÃO pode ser reatribuída
val pi = 3.14159
pi = 3.14  // ❌ ERRO!

val cpf = "123.456.789-00"
cpf = "000"  // ❌ ERRO!

// Mas objetos podem mudar internamente
val lista = mutableListOf(1, 2, 3)
lista.add(4)  // ✅ OK
var - Mutável 

// Pode ser reatribuída
var nome = "Maria"
nome = "Maria Silva"  // ✅ OK

var contador = 0
contador++  // ✅ OK
contador = 100  // ✅ OK

var cidade = "Recife"
cidade = "São Paulo"  // ✅ OK
⚠️ Regra de ouro: Sempre use val por padrão. Use var apenas quando realmente precisar alterar o valor.
