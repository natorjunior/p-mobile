class Calculadora {
    var resultado: Double = 0.0
    
    constructor(valor: Double = 0.0) {
        resultado = valor
    }
    
    fun somar(valor: Double) {
        resultado += valor
    }
    
    fun multiplicar(valor: Double) {
        resultado *= valor
    }
    
    fun obter(): Double {
        return resultado
    }
}

🧮 Usando a Calculadora


fun main() {
    val calc = Calculadora(10.0)
    calc.somar(5.0)
    calc.multiplicar(2.0)
    
    println("Resultado: ${calc.obter()}")  // 30.0
}

Resultado Final

Calculadora(10.0) → somar(5.0) → multiplicar(2.0) = 30.0

🎨 Tela (Compose) 

@Composable
fun RegistroScreen(vm: RegistroVM) {
    var nome by remember { 
        mutableStateOf("") 
    }
    
    TextField(value = nome,
        onValueChange = { nome = it })
    
    Button(onClick = { 
        vm.salvarUsuario(nome)
    }) {
        Text("Cadastrar")
    }
}
⚙️ Lógica (ViewModel) 

class RegistroVM(
    val repo: UsuarioRepo
) : ViewModel() {

    fun salvarUsuario(nome: String) {
        if (nome.isBlank()) return
        
        val user = Usuario(
            nome = nome,
            data = LocalDate.now()
        )
        repo.salvar(user)
    }
}
📌 Observe: A tela só sabe chamar o ViewModel. O ViewModel só sabe chamar o Repository. Ninguém faz tudo sozinho!

🔁 for: Intervalo simples

🔁 for: Intervalo exclusivo


for (i in 1 until 5) {

🔁 for: Ordem decrescente

    println(i)
}
// Saída: 5 4 3 2 1

🔁 for: Com passo (step)


for (i in 1..10 step 2) {
    println(i)
}
// Saída: 1 3 5 7 9

🔁 for: Iterando em lista

    println("Olá, $nome!")

🔁 while: Contadora


var contador = 1

while (contador <= 5) {
    println("Contagem: $contador")
    contador++
}

🔁 while: Com condição complexa


var numero = 10

while (numero > 0) {
    println(numero)
    numero -= 2

🔁 do-while: Executa primeiro


var tentativa = 1

do {
    println("Tentativa #$tentativa")
    tentativa++

🔁 break: Sair do loop


for (i in 1..10) {
    if (i == 5) {
        break  // Sai do loop
    }
    println(i)
}
// Saída: 1 2 3 4

🔁 continue: Pula iteração


for (i in 1..5) {
    if (i == 3) {
        continue  // Pula para próxima
    }
    println(i)
}
// Saída: 1 2 4 5

🧪 Mini Desafio Kotlin

5 minutos

🧪 Mini Desafio

Dada uma lista de números, gere outra lista contendo apenas os pares, elevados ao quadrado, ordenados de forma crescente. 


val nums = listOf(9, 2, 4, 7, 6, 1, 8)
// TODO: resultado esperado -> [4, 16, 36, 64]

✅ Resolução


val nums = listOf(9, 2, 4, 7, 6, 1, 8)

val resultado = nums
  .filter { it % 2 == 0 }
  .map { it * it }
  .sorted()

println(resultado)  // [4, 16, 36, 64]
Resumo: filter (seleciona) → map (transforma) → sorted (ordena).
