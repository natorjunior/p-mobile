📦 Variáveis: val vs var

val - Imutável 

// NÃO pode ser reatribuída
val pi = 3.14159
pi = 3.14  // ❌ ERRO!

val cpf = "123.456.789-00"
cpf = "000"  // ❌ ERRO!

// Mas objetos podem mudar internamente
val lista = mutableListOf(1, 2, 3)
lista.add(4)  // ✅ OK
var - Mutável 

// Pode ser reatribuída
var nome = "Maria"
nome = "Maria Silva"  // ✅ OK

var contador = 0
contador++  // ✅ OK
contador = 100  // ✅ OK

var cidade = "Recife"
cidade = "São Paulo"  // ✅ OK
⚠️ Regra de ouro: Sempre use val por padrão. Use var apenas quando realmente precisar alterar o valor.

🛡️ Null Safety - Adeus NullPointerException!


// 📌 Por padrão, variáveis NÃO podem ser null
var nome: String = "João"
nome = null  // ❌ ERRO DE COMPILAÇÃO!

// 📌 Para permitir null, use ?
var email: String? = null  // ✅ OK
email = "joao@email.com"   // ✅ OK

// 📌 Acesso seguro com ?.
println(email?.length)     // 15 (ou null se email for null)
println(email?.uppercase())

// 📌 Elvis operator ?: (valor padrão)
val tamanho = email?.length ?: 0
println("Tamanho: $tamanho")

// 📌 Safe call chain
data class Endereco(val rua: String?)
data class Pessoa(val endereco: Endereco?)

val pessoa: Pessoa? = null
println(pessoa?.endereco?.rua ?: "Não informado")

🔒 Mais Null Safety


// 📌 let - Executar bloco se não for null
val email: String? = "teste@email.com"

email?.let { e ->
    println("Email válido: $e")
    enviarEmail(e)
}

// 📌 Versão compacta com it
email?.let {
    println("Email: $it")
}

// 📌 !! - Force unwrap (CUIDADO! Pode crashar)
val tamanho = email!!.length  // Crash se null!

// 📌 Verificação com if
if (email != null) {
    // Smart cast: email é String aqui (não String?)
    println(email.length)
}

// 📌 require e check
fun processar(nome: String?) {
    requireNotNull(nome) { "Nome é obrigatório!" }
    // nome é String aqui (não String?)
    println(nome.uppercase())
}
⚠️ Evite !! - Se precisar usar, provavelmente há um design melhor.

🔄 State - Gerenciando Estado


@Composable
fun Contador() {
    // 👇 remember + mutableStateOf = estado reativo
    var contador by remember { mutableStateOf(0) }
    
    Column(
        modifier = Modifier.padding(16.dp),
        horizontalAlignment = Alignment.CenterHorizontally
    ) {
        Text(
            text = "Contador: $contador",
            style = MaterialTheme.typography.headlineLarge
        )
        
        Spacer(modifier = Modifier.height(16.dp))
        
        Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            Button(onClick = { contador-- }) {
                Text("−")
            }
            Button(onClick = { contador++ }) {
                Text("+")
            }
        }
    }
}
💡 remember mantém o valor entre recomposições. mutableStateOf torna o valor observável.

🧭 Navegação entre Telas


// 📌 Definir rotas
sealed class Rota(val nome: String) {
    object Home : Rota("home")
    object Perfil : Rota("perfil")
    object Detalhes : Rota("detalhes/{id}") // Com parâmetro
}

@Composable
fun AppNavigation() {
    val navController = rememberNavController()
    
    NavHost(navController = navController, startDestination = Rota.Home.nome) {
        composable(Rota.Home.nome) {
            TelaHome(
                onNavigateToPerfil = { navController.navigate(Rota.Perfil.nome) },
                onNavigateToDetalhes = { id -> 
                    navController.navigate("detalhes/$id") 
                }
            )
        }
        composable(Rota.Perfil.nome) {
            TelaPerfil(onVoltar = { navController.popBackStack() })
        }
        composable(
            route = Rota.Detalhes.nome,
            arguments = listOf(navArgument("id") { type = NavType.IntType })
        ) { backStackEntry ->
            val id = backStackEntry.arguments?.getInt("id") ?: 0
            TelaDetalhes(id = id)
        }
    }
}

O sistema tradicional do Android

📱 XML Views - Quando usar?

✅ Use XML quando:

  • Projeto legado existente
  • Equipe ainda não domina Compose
  • Funcionalidades não disponíveis no Compose
  • Suporte a Android < 5.0

✅ Use Compose quando:

  • Projeto novo
  • UI complexa e dinâmica
  • Melhor produtividade
  • Código mais conciso
💡 Na prática: Muitos apps usam os dois! Compose pode incorporar views XML e vice-versa com AndroidView e ComposeView.

        val binding = ItemTarefaBinding.inflate(

            root.setOnClickListener { onItemClick(tarefa) }
        }
    }

}

class TarefasViewModel : ViewModel() {
    
    private val _tarefas = mutableStateListOf<Tarefa>()
    val tarefas: List<Tarefa> = _tarefas
    
    var novaTarefa by mutableStateOf("")
    var novaDescricao by mutableStateOf("")
    
    fun adicionar() {
        if (novaTarefa.isBlank()) return
        _tarefas.add(Tarefa(titulo = novaTarefa, descricao = novaDescricao))
        novaTarefa = ""
        novaDescricao = ""
    }
    
    fun remover(tarefa: Tarefa) {
        _tarefas.remove(tarefa)
    }
    
    fun alternarConclusao(tarefa: Tarefa) {
        val index = _tarefas.indexOf(tarefa)
        if (index != -1) {
            _tarefas[index] = tarefa.copy(concluida = !tarefa.concluida)
        }
    }
}

            contentPadding = PaddingValues(16.dp),
            verticalArrangement = Arrangement.spacedBy(8.dp)

    }
}

@Composable
fun TarefaCard(
    tarefa: Tarefa,
    onToggle: (Tarefa) -> Unit,
    onRemover: (Tarefa) -> Unit
) {
    Card(
        modifier = Modifier.fillMaxWidth(),
        colors = CardDefaults.cardColors(
            containerColor = if (tarefa.concluida) 
                Color(0xFFE8F5E9) else MaterialTheme.colorScheme.surface
        )
    ) {
        Row(
            modifier = Modifier.padding(16.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            Checkbox(
                checked = tarefa.concluida,
                onCheckedChange = { onToggle(tarefa) }
            )
            Column(modifier = Modifier.weight(1f)) {
                Text(
                    text = tarefa.titulo,
                    style = MaterialTheme.typography.titleMedium,
                    textDecoration = if (tarefa.concluida) 
                        TextDecoration.LineThrough else null
                )
                Text(tarefa.dataFormatada, style = MaterialTheme.typography.bodySmall)
            }
            IconButton(onClick = { onRemover(tarefa) }) {
                Icon(Icons.Default.Delete, "Remover", tint = Color.Red)
            }
        }
    }
}

@Composable
fun NovaTarefaDialog(
    viewModel: TarefasViewModel,
    onDismiss: () -> Unit
) {
    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text("Nova Tarefa") },
        text = {
            Column {
                OutlinedTextField(
                    value = viewModel.novaTarefa,
                    onValueChange = { viewModel.novaTarefa = it },
                    label = { Text("Título") },
                    modifier = Modifier.fillMaxWidth()
                )
                Spacer(Modifier.height(8.dp))
                OutlinedTextField(
                    value = viewModel.novaDescricao,
                    onValueChange = { viewModel.novaDescricao = it },
                    label = { Text("Descrição (opcional)") },
                    modifier = Modifier.fillMaxWidth()
                )
            }
        },
        confirmButton = {
            Button(
                onClick = { viewModel.adicionar(); onDismiss() },
                enabled = viewModel.novaTarefa.isNotBlank()
            ) { Text("Salvar") }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) { Text("Cancelar") }
        }
    )
}
