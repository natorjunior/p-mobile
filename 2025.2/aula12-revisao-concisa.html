🪝 Hooks

Gerenciamento de estado e efeitos

useState - Estado Local

Função: Armazena e atualiza dados dinâmicos do componente

Sintaxe: const [valor, setValor] = useState(inicial)


import { useState } from 'react';
import { View, Text, Button } from 'react-native';

export default function Contador() {
  const [count, setCount] = useState(0);

  return (
    <View>
      <Text>Contador: {count}</Text>
      <Button title="+" onPress={() => setCount(count + 1)} />
      <Button title="-" onPress={() => setCount(count - 1)} />
      <Button title="Reset" onPress={() => setCount(0)} />
    </View>
  );
}

useEffect - Efeitos Colaterais

Função: Executa código em momentos específicos do ciclo de vida

✅ [] - Uma vez (mount)

✅ Sem array - Sempre (update)

✅ [dep] - Quando dep muda

✅ return () => {} - Cleanup


import { useState, useEffect } from 'react';
import { Text } from 'react-native';

export default function Timer() {
  const [segundos, setSegundos] = useState(0);

  useEffect(() => {
    const interval = setInterval(() => {
      setSegundos(s => s + 1);
    }, 1000);

    return () => clearInterval(interval); // Cleanup
  }, []); // Executa uma vez

  return <Text>Tempo: {segundos}s</Text>;
}

Combinando useState + useEffect


import { useState, useEffect } from 'react';
import { View, Text, TextInput } from 'react-native';

export default function BuscaApp() {
  const [busca, setBusca] = useState('');
  const [resultados, setResultados] = useState([]);

  useEffect(() => {
    if (busca.length > 2) {
      // Simula busca com delay
      const timer = setTimeout(() => {
        const filtrados = dados.filter(item => 
          item.toLowerCase().includes(busca.toLowerCase())
        );
        setResultados(filtrados);
      }, 300);

      return () => clearTimeout(timer); // Cleanup
    }
  }, [busca]); // Executa quando 'busca' muda

  return (
    <View>
      <TextInput 
        placeholder="Buscar..."
        value={busca}
        onChangeText={setBusca}
      />
      <Text>Resultados: {resultados.length}</Text>
    </View>
  );
}

🌐 Consumo de APIs

fetch e integração de dados

fetch - Básico

Função: Buscar dados de APIs REST

Métodos: GET, POST, PUT, DELETE


import { useState, useEffect } from 'react';
import { View, Text, FlatList, ActivityIndicator } from 'react-native';

export default function ListaUsuarios() {
  const [usuarios, setUsuarios] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch('https://jsonplaceholder.typicode.com/users')
      .then(response => response.json())
      .then(data => {
        setUsuarios(data);
        setLoading(false);
      })
      .catch(error => console.error(error));
  }, []);

  if (loading) return <ActivityIndicator size="large" />;

  return (
    <FlatList
      data={usuarios}
      keyExtractor={item => item.id.toString()}
      renderItem={({ item }) => <Text>{item.name}</Text>}
    />
  );
}

Tratamento de Erros


import { useState, useEffect } from 'react';
import { View, Text, ActivityIndicator } from 'react-native';

export default function AppComErros() {
  const [dados, setDados] = useState(null);
  const [loading, setLoading] = useState(true);
  const [erro, setErro] = useState(null);

  useEffect(() => {
    fetch('https://api.exemplo.com/dados')
      .then(response => {
        if (!response.ok) throw new Error('Erro na requisição');
        return response.json();
      })
      .then(data => {
        setDados(data);
        setLoading(false);
      })
      .catch(error => {
        setErro(error.message);
        setLoading(false);
      });
  }, []);

  if (loading) return <ActivityIndicator />;
  if (erro) return <Text>Erro: {erro}</Text>;
  return <Text>{JSON.stringify(dados)}</Text>;
}

💾 AsyncStorage

Persistência de dados local

AsyncStorage - Conceito

O que é? Sistema de armazenamento chave-valor local

Uso: Salvar preferências, dados offline, cache

Importante: Dados são strings (usar JSON para objetos)


# Instalação
npm install @react-native-async-storage/async-storage

Operações Básicas


import AsyncStorage from '@react-native-async-storage/async-storage';

// SALVAR
const salvarDado = async (chave, valor) => {
  try {
    await AsyncStorage.setItem(chave, valor);
    console.log('Salvo!');
  } catch (error) {
    console.error('Erro ao salvar:', error);
  }
};

// LER
const lerDado = async (chave) => {
  try {
    const valor = await AsyncStorage.getItem(chave);
    return valor !== null ? valor : 'Não encontrado';
  } catch (error) {
    console.error('Erro ao ler:', error);
  }
};

// REMOVER
const removerDado = async (chave) => {
  try {
    await AsyncStorage.removeItem(chave);
  } catch (error) {
    console.error('Erro ao remover:', error);
  }
};

Salvar Objetos JSON


import { useState, useEffect } from 'react';
import AsyncStorage from '@react-native-async-storage/async-storage';

export default function AppStorage() {
  const [usuario, setUsuario] = useState({ nome: '', email: '' });

  // SALVAR objeto
  const salvarUsuario = async () => {
    try {
      const jsonValue = JSON.stringify(usuario);
      await AsyncStorage.setItem('usuario', jsonValue);
      alert('Salvo!');
    } catch (error) {
      console.error(error);
    }
  };

  // CARREGAR objeto
  const carregarUsuario = async () => {
    try {
      const jsonValue = await AsyncStorage.getItem('usuario');
      if (jsonValue !== null) {
        setUsuario(JSON.parse(jsonValue));
      }
    } catch (error) {
      console.error(error);
    }
  };

  useEffect(() => {
    carregarUsuario(); // Carrega ao iniciar
  }, []);

  return (
    <Button title="Salvar" onPress={salvarUsuario} />
  );
}

Exemplo Completo: Lista de Tarefas


import { useState, useEffect } from 'react';
import { View, TextInput, Button, FlatList, Text } from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';

export default function TodoApp() {
  const [tarefa, setTarefa] = useState('');
  const [tarefas, setTarefas] = useState([]);

  // Carregar tarefas ao iniciar
  useEffect(() => {
    carregarTarefas();
  }, []);

  // Salvar sempre que tarefas mudar
  useEffect(() => {
    salvarTarefas();
  }, [tarefas]);

  const salvarTarefas = async () => {
    try {
      await AsyncStorage.setItem('tarefas', JSON.stringify(tarefas));
    } catch (error) {
      console.error(error);
    }
  };

  const carregarTarefas = async () => {
    try {
      const data = await AsyncStorage.getItem('tarefas');
      if (data) setTarefas(JSON.parse(data));
    } catch (error) {
      console.error(error);
    }
  };

  const adicionarTarefa = () => {
    if (tarefa.trim()) {
      setTarefas([...tarefas, { id: Date.now().toString(), texto: tarefa }]);
      setTarefa('');
    }
  };

  return (
    <View style={{ padding: 20 }}>
      <TextInput
        value={tarefa}
        onChangeText={setTarefa}
        placeholder="Nova tarefa"
      />
      <Button title="Adicionar" onPress={adicionarTarefa} />
      
      <FlatList
        data={tarefas}
        keyExtractor={item => item.id}
        renderItem={({ item }) => <Text>{item.texto}</Text>}
      />
    </View>
  );
}

HomeScreen - Lista com API


import { useState, useEffect } from 'react';
import { View, FlatList, ActivityIndicator } from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';

export default function HomeScreen({ navigation }) {
  const [usuarios, setUsuarios] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    carregarUsuarios();
  }, []);

  const carregarUsuarios = async () => {
    try {
      const response = await fetch('https://jsonplaceholder.typicode.com/users');
      const data = await response.json();
      setUsuarios(data);
    } catch (error) {
      console.error(error);
    } finally {
      setLoading(false);
    }
  };

  const adicionarFavorito = async (usuario) => {
    try {
      const favs = await AsyncStorage.getItem('favoritos');
      const favoritos = favs ? JSON.parse(favs) : [];
      favoritos.push(usuario);
      await AsyncStorage.setItem('favoritos', JSON.stringify(favoritos));
      alert('Adicionado aos favoritos!');
    } catch (error) {
      console.error(error);
    }
  };

  if (loading) return <ActivityIndicator size="large" />;

  return (
    <FlatList
      data={usuarios}
      keyExtractor={item => item.id.toString()}
      renderItem={({ item }) => (
        <View style={{ padding: 15, borderBottomWidth: 1 }}>
          <Text onPress={() => navigation.navigate('Detalhes', { usuario: item })}>
            {item.name}
          </Text>
          <Button title="♥ Favoritar" onPress={() => adicionarFavorito(item)} />
        </View>
      )}
    />
  );
}

FavoritosScreen - Storage


import { useState, useEffect } from 'react';
import { View, FlatList, Text, Button } from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';

export default function FavoritosScreen() {
  const [favoritos, setFavoritos] = useState([]);

  useEffect(() => {
    carregarFavoritos();
  }, []);

  const carregarFavoritos = async () => {
    try {
      const data = await AsyncStorage.getItem('favoritos');
      if (data) setFavoritos(JSON.parse(data));
    } catch (error) {
      console.error(error);
    }
  };

  const removerFavorito = async (id) => {
    try {
      const novos = favoritos.filter(item => item.id !== id);
      setFavoritos(novos);
      await AsyncStorage.setItem('favoritos', JSON.stringify(novos));
    } catch (error) {
      console.error(error);
    }
  };

  return (
    <View>
      {favoritos.length === 0 ? (
        <Text>Nenhum favorito</Text>
      ) : (
        <FlatList
          data={favoritos}
          keyExtractor={item => item.id.toString()}
          renderItem={({ item }) => (
            <View style={{ padding: 15 }}>
              <Text>{item.name}</Text>
              <Button title="Remover" onPress={() => removerFavorito(item.id)} />
            </View>
          )}
        />
      )}
    </View>
  );
}
